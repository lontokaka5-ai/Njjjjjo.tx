    async function humanClick(el) {
        if (!el) return false;
        try {
            const r = el.getBoundingClientRect();
            const tx = r.left + r.width / 2, ty = r.top + r.height / 2;
            const pid = Math.floor(Math.random() * 1e6);

                const base = { bubbles: true, cancelable: true, composed: true, view: window,
                    clientX: x, clientY: y, screenX: x, screenY: y, button: 0, buttons: buttons || 0 };
                const ptr = { ...base, pointerId: pid, pointerType: 'mouse', isPrimary: true,
                    width: 1, height: 1, pressure: type === 'pointerdown' ? 0.5 : 0 };
                try { el.dispatchEvent(new PointerEvent(type, ptr)); } catch (e) {}
                try { document.dispatchEvent(new PointerEvent(type, ptr)); } catch (e) {}
                const mtype = 'mouse' + type.slice(7);
                if (['mousemove','mousedown','mouseup'].includes(mtype)) {
                    try { el.dispatchEvent(new MouseEvent(mtype, base)); } catch (e) {}
                    try { document.dispatchEvent(new MouseEvent(mtype, base)); } catch (e) {}
                }
            };
            for (let i = 1; i <= steps; i++) {
                const t = i / steps;
                const x = cx + (tx - cx) * t + (Math.random() - 0.5) * CONFIG.human.mouseJitter;
                const y = cy + (ty - cy) * t + (Math.random() - 0.5) * CONFIG.human.mouseJitter;
                dispatch('pointermove', x, y, 0);
                await sleep(rand(CONFIG.human.mouseStepDelay));
            }
            await sleep(rand(CONFIG.human.preClick));
            if (Math.random() < CONFIG.human.focusJitterChance) await sleep(rand(CONFIG.human.focusJitter));
            dispatch('pointerdown', tx, ty, 1);
            await sleep(rand(CONFIG.human.clickHold));
            dispatch('pointerup', tx, ty, 0);
            try { el.click(); } catch (e) {}
            await sleep(rand(CONFIG.human.postClick));
            return true;
        } catch (e) { return false; }
    }

    /* ========== SWIPE / NEXT ========== */
    async function simulateSwipeUp(target) {
        const el = target || findActiveVideo() || document.body;
        const rect = el.getBoundingClientRect ? el.getBoundingClientRect() : { left: 0, top: 0, width: innerWidth, height: innerHeight };
        const cx = rect.left + rect.width / 2;
        const yStart = Math.min(rect.top + rect.height * 0.78, innerHeight - 80);
        const yEnd = Math.max(rect.top + rect.height * 0.12, 120);
        const steps = 7, pid = 1 + Math.floor(Math.random() * 1000);
        const touchEvt = (type, x, y) => {
            try {
                const t = new Touch({ identifier: pid, target: el, clientX: x, clientY: y, radiusX: 2, radiusY: 2, force: 0.5 });
                const init = { bubbles: true, cancelable: true, composed: true, view: window,
                    touches: type === 'touchend' ? [] : [t], targetTouches: type === 'touchend' ? [] : [t], changedTouches: [t] };
                el.dispatchEvent(new TouchEvent(type, init));
                document.dispatchEvent(new TouchEvent(type, init));
            } catch (e) {}
        };
        const pointerEvt = (type, x, y) => {
            try {
                const base = { bubbles: true, cancelable: true, composed: true, view: window, clientX: x, clientY: y,
                    pointerId: pid, pointerType: 'touch', isPrimary: true, width: 2, height: 2,
                    pressure: type === 'pointerup' ? 0 : 0.5, buttons: type === 'pointerup' ? 0 : 1 };
                el.dispatchEvent(new PointerEvent(type, base));
                document.dispatchEvent(new PointerEvent(type, base));
            } catch (e) {}
        };
        pointerEvt('pointerdown', cx, yStart); touchEvt('touchstart', cx, yStart);
        await sleep(25);
        for (let i = 1; i <= steps; i++) {
            const y = yStart + (yEnd - yStart) * (i / steps);
            pointerEvt('pointermove', cx, y); touchEvt('touchmove', cx, y);
            await sleep(14 + Math.random() * 20);
        }
        pointerEvt('pointerup', cx, yEnd); touchEvt('touchend', cx, yEnd);
    }

    async function doNext() {
        if (!S.running || S.nextInProgress) return;
        S.nextInProgress = true;
        try {
            SafeTimer.clearAll();
            const dlg = findShareDialog();
            if (dlg) {
                const closeBtn = findShareCloseBtn();
                if (closeBtn) await humanClick(closeBtn); else fireEsc();
                await sleep(300);
            }
            const sp = CONFIG.speeds[CONFIG.speed] || CONFIG.speeds.normal;
            await sleep(hDelay(sp.next));
            if (!S.running) return;
            const tgt = findScrollTarget();
            if (tgt && tgt.scrollBy) {
                const h = tgt.clientHeight;
                try { tgt.scrollBy({ top: h, behavior: 'smooth' }); } catch (e) { tgt.scrollTop += h; }
                log('next: scroll', '#0f0');
            }
            try { document.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowDown', code: 'ArrowDown', keyCode: 40, bubbles: true })); } catch (e) {}
            simulateSwipeUp(findActiveVideo());
            log('next: swipe', '#0f0');

            S.videos++;
            S.videoId = null; S.videoStart = 0; S.lastCT = 0;
            S.endFiredFor = null; S.firedFor = {}; S.actionsThisVideo = 0;
            updateUI();
            if (S.videos % CONFIG.longPauseEvery === 0) {
                const p = hDelay(CONFIG.longPause);
                log(`nghỉ dài ${(p / 1000) | 0}s`, '#fa0');
                toast(`Nghỉ dài ${(p / 1000) | 0}s`, 'ok');
                await sleep(p);
            } else await sleep(hDelay(sp.between));
        } finally { S.nextInProgress = false; }
    }

    /* ========== ATTACH / POLL ========== */
    function attachVideo() {
        const v = findActiveVideo();
        if (!v) {
            S.attachFailCount++;
            if (S.attachFailCount >= 5) recover('attach fail x5');
            return;
        }
        if (v === S.videoEl && S.videoId) return;
        if (S.videoEl && S.videoEl !== v) {
            try {
                S.videoEl.removeEventListener('ended', onEnded);
                S.videoEl.removeEventListener('pause', onPause);
            } catch (e) {}
        }
        SafeTimer.clearAll();
        S.videoEl = v;
        const srcKey = (v.currentSrc || v.src || '').split('?')[0].slice(-60);
        S.videoId = `${srcKey}|${Math.floor(Date.now() / 1000)}`;
        S.videoStart = Date.now();
        S.lastCT = v.currentTime || 0;
        S.endFiredFor = null; S.firedFor = {}; S.actionsThisVideo = 0;
        S.attachFailCount = 0;
        try {
            v.addEventListener('ended', onEnded);
            v.addEventListener('pause', onPause);
        } catch (e) {}
        if (CONFIG.autoUnmute) setTimeout(doUnmute, 200);
        log(`video mới (${v.duration ? v.duration.toFixed(1) + 's' : '?'})`, '#0af');

        const reaction = CONFIG.human.enabled ? hDelay([500, 1600]) : 0;
        if (CONFIG.autoLike)   setTimeout(scheduleLike,   reaction);
        if (CONFIG.autoFollow) setTimeout(scheduleFollow, reaction);
        if (CONFIG.autoShare)  setTimeout(scheduleShare,  reaction);
        if (CONFIG.autoSave)   setTimeout(scheduleSave,   reaction);
    }

    function pollVideo() {
        if (!S.running) return;
        const v = S.videoEl;
        if (!v || !v.isConnected) { attachVideo(); return; }
        try {
            const ct = v.currentTime, dur = v.duration;
            if (ct + 1.5 < S.lastCT) { log('video mới (ct reset)', '#0af'); S.videoId = null; attachVideo(); return; }
            S.lastCT = ct;
            if (dur && ct >= dur - CONFIG.endThreshold) {
                if (S.endFiredFor !== S.videoId) {
                    S.endFiredFor = S.videoId;
                    log(`kết thúc @${ct.toFixed(1)}/${dur.toFixed(1)}`, '#0f0');
                    onEnded();
                }
                return;
            }
            if (S.videoStart && (Date.now() - S.videoStart) / 1000 > CONFIG.maxWatchSec) {
                if (S.endFiredFor !== S.videoId) {
                    S.endFiredFor = S.videoId;
                    log(`vượt ${CONFIG.maxWatchSec}s — next`, '#fa0');
                    onEnded();
                }
            }
        } catch (e) {}
    }
    function onEnded() {
        if (!S.running) return;
        SafeTimer.clearAll();
        S.watchSec += (Date.now() - S.videoStart) / 1000;
        if (CONFIG.autoNext) doNext();
    }
    function onPause() {
        if (!S.running || !S.videoEl) return;
        try {
            const { currentTime, duration } = S.videoEl;
            if (duration && currentTime >= duration - CONFIG.endThreshold) return;
            if (CONFIG.autoNext) doNext();
        } catch (e) {}
    }

    /* ========== WATCHDOG ========== */
    function startWatchdog() {
        if (S.watchdog) clearInterval(S.watchdog);
        S.watchdog = setInterval(() => {
            if (!S.running) return;
            try {
                // video còn sống?
                if (!S.videoEl || !S.videoEl.isConnected) {
                    log('watchdog: video mất', '#fa0');
                    attachVideo();
                    return;
                }
                // treo? currentTime không đổi trong 30s + không pause
                if (Date.now() - S.videoStart > 30000 && S.lastCT === S.videoEl.currentTime && !S.videoEl.paused && !S.endFiredFor) {
                    log('watchdog: video treo', '#fa0');
                    recover('video stuck');
                }
            } catch (e) { S.crashCount++; }
        }, 5000);
    }

    /* ========== MUTATION OBSERVER ========== */
    function startObserver() {
        if (S.observer) try { S.observer.disconnect(); } catch (e) {}
        try {
            S.observer = new MutationObserver(() => {
                if (S.running && !S.videoEl) attachVideo();
            });
            S.observer.observe(document.body, { childList: true, subtree: true });
        } catch (e) {}
    }

    /* ========== CONTROL ========== */
    function start() {
        if (S.running) return;
        S.running = true; S.startTs = Date.now(); S.attachFailCount = 0; S.crashCount = 0;
        log('ON', '#0f0'); toast('▶ Bắt đầu', 'ok');
        attachVideo(); updateUI();
    }
    function stop() {
        if (!S.running) return;
        S.running = false;
        SafeTimer.clearAll();
        if (S.videoEl) {
            try {
                S.videoEl.removeEventListener('ended', onEnded);
                S.videoEl.removeEventListener('pause', onPause);
            } catch (e) {}
        }
        log(`OFF | ${S.videos} videos`, '#f44');
        toast(`■ Dừng · ${S.videos} videos`, 'err');
        updateUI();
    }
    function resetStats() {
        S.videos = S.likes = S.follows = S.shares = S.saves = 0;
        S.watchSec = 0; S.startTs = Date.now();
        S.logBuf = []; renderLog(); updateUI();
        toast('↻ Đã reset', 'ok');
    }
    function destroy() {
        try { stop(); } catch (e) {}
        if (S.heartbeat) clearInterval(S.heartbeat);
        if (S.watchdog) clearInterval(S.watchdog);
        if (S.uiTick) clearInterval(S.uiTick);
        if (S.observer) try { S.observer.disconnect(); } catch (e) {}
        SafeTimer.clearAll();
        const h = document.getElementById('__tt_host');
        if (h) h.remove();
        delete window.__TT;
    }

        bindToggle(rUn,     'autoUnmute');

        for (const k in spBtns) {
            spBtns[k].addEventListener('click', e => {
                e.stopPropagation();
                CONFIG.speed = k;
                for (const kk in spBtns) spBtns[kk].classList.toggle('on', kk === k);
                toast(`Tốc độ: ${k}`, 'ok'); saveCfg();
            });
        }

        hmSw.addEventListener('click', e => {
            e.stopPropagation();
            CONFIG.human.enabled = !CONFIG.human.enabled;
            hmSw.classList.toggle('on', CONFIG.human.enabled);
            hmRow.classList.toggle('active', CONFIG.human.enabled);
            toast(`Human: ${CONFIG.human.enabled ? 'ON' : 'OFF'}`, 'ok');
            updateUI(); saveCfg();
        });

        resetCfgBtn.addEventListener('click', e => {
            e.stopPropagation();
            for (const k in DEFAULTS) CONFIG[k] = DEFAULTS[k].slice();
            CONFIG.human.enabled = true;
            hmSw.classList.toggle('on', true);
            hmRow.classList.toggle('active', true);
            drLike.refresh(); drFollow.refresh(); drShare.refresh(); drSave.refresh();
            toast('↻ Đã reset cấu hình', 'ok'); saveCfg();
        });

        goBtn.addEventListener('click', e => { e.stopPropagation(); S.running ? stop() : start(); });
        skipBtn.addEventListener('click', e => { e.stopPropagation(); if (S.running) doNext(); else toast('Chưa chạy', 'err'); });
        resetBtn.addEventListener('click', e => { e.stopPropagation(); resetStats(); });

        // Panel drag
        let pDrag = null;
        hdr.addEventListener('pointerdown', e => {
            if (e.target === hdrMin) return;
            e.stopPropagation();
            const r = panel.getBoundingClientRect();
            pDrag = { id: e.pointerId, dx: e.clientX - r.left, dy: e.clientY - r.top };
            panel.style.transition = 'none';
        });
        document.addEventListener('pointermove', e => {
            if (!pDrag || e.pointerId !== pDrag.id) return;
            let x = e.clientX - pDrag.dx, y = e.clientY - pDrag.dy;
            x = Math.max(6, Math.min(innerWidth - panel.offsetWidth - 6, x));
            y = Math.max(6, Math.min(innerHeight - panel.offsetHeight - 6, y));
            panel.style.left = x + 'px'; panel.style.top = y + 'px';
            panel.style.right = 'auto'; panel.style.bottom = 'auto';
        }, true);
        document.addEventListener('pointerup', e => {
            if (!pDrag || e.pointerId !== pDrag.id) return;
            pDrag = null; panel.style.transition = '';
        }, true);

        S.ui = {
            fab, fabBadge, panel, overlay, goBtn, dot, stxt, toast: toastEl, logBox, status,
            vs: $id('v-s'), vw: $id('v-w'), vl: $id('v-l'), vu: $id('v-u'),
            vsh: $id('v-sh'), vsv: $id('v-sv'),
            drLike, drFollow, drShare, drSave, hmSw, hmRow,
            openPanel, closePanel,
        };
    }

    /* ========== CONFIG PERSIST ========== */
    function saveCfg() {
        try {
            const out = {
                v: 6,
                likeDelay: CONFIG.likeDelay, followDelay: CONFIG.followDelay,
                shareDelay: CONFIG.shareDelay, saveDelay: CONFIG.saveDelay,
                speed: CONFIG.speed, humanEnabled: CONFIG.human.enabled,
                autoNext: CONFIG.autoNext, autoLike: CONFIG.autoLike,
                autoFollow: CONFIG.autoFollow, autoShare: CONFIG.autoShare,
                autoSave: CONFIG.autoSave, autoUnmute: CONFIG.autoUnmute,
            };
            localStorage.setItem(STORE_KEY, JSON.stringify(out));
        } catch (e) {}
    }
    function loadCfg() {
        try {
            const raw = localStorage.getItem(STORE_KEY);
            if (!raw) return;
            const o = JSON.parse(raw);
            if (o.v !== 6) return; // version mismatch
            if (o.likeDelay)   CONFIG.likeDelay = o.likeDelay;
            if (o.followDelay) CONFIG.followDelay = o.followDelay;
            if (o.shareDelay)  CONFIG.shareDelay = o.shareDelay;
            if (o.saveDelay)   CONFIG.saveDelay = o.saveDelay;
            if (o.speed && CONFIG.speeds[o.speed]) CONFIG.speed = o.speed;
            if (typeof o.humanEnabled === 'boolean') CONFIG.human.enabled = o.humanEnabled;
            if (typeof o.autoNext === 'boolean')   CONFIG.autoNext = o.autoNext;
            if (typeof o.autoLike === 'boolean')   CONFIG.autoLike = o.autoLike;
            if (typeof o.autoFollow === 'boolean') CONFIG.autoFollow = o.autoFollow;
            if (typeof o.autoShare === 'boolean')  CONFIG.autoShare = o.autoShare;
            if (typeof o.autoSave === 'boolean')   CONFIG.autoSave = o.autoSave;
            if (typeof o.autoUnmute === 'boolean') CONFIG.autoUnmute = o.autoUnmute;
        } catch (e) {}
    }

    /* ========== UPDATE UI ========== */
    let lastVals = { s: 0, l: 0, u: 0, sh: 0, sv: 0 };
    function pulseCell(el) { if (!el) return; el.classList.remove('pulse'); void el.offsetWidth; el.classList.add('pulse'); }
    function updateUI() {
        if (!S.ui) return;
        try {
            const { fab, fabBadge, goBtn, dot, stxt, status, vs, vw, vl, vu, vsh, vsv } = S.ui;
            fab.classList.toggle('on', S.running);
            if (S.videos > 0) { fabBadge.style.display = 'flex'; fabBadge.textContent = S.videos > 999 ? '999+' : String(S.videos); }
            else fabBadge.style.display = 'none';
            goBtn.textContent = S.running ? 'Dừng' : 'Bắt đầu';
            goBtn.classList.toggle('stop', S.running);
            dot.classList.toggle('on', S.running);
            status.classList.toggle('on', S.running);

            stxt.textContent = '';
            const b = document.createElement('b');
            b.textContent = S.running ? 'Đang chạy' : 'Đang dừng';
            stxt.appendChild(b);
            if (S.running) {
                const chip = document.createElement('span');
                chip.className = 'chip';
                chip.textContent = CONFIG.human.enabled ? `human·${CONFIG.speed}` : CONFIG.speed;
                stxt.appendChild(chip);
            }

            if (S.videos  !== lastVals.s)  { pulseCell(vs.parentElement);  lastVals.s  = S.videos;  }
            if (S.likes   !== lastVals.l)  { pulseCell(vl.parentElement);  lastVals.l  = S.likes;   }
            if (S.follows !== lastVals.u)  { pulseCell(vu.parentElement);  lastVals.u  = S.follows; }
            if (S.shares  !== lastVals.sh) { pulseCell(vsh.parentElement); lastVals.sh = S.shares;  }
            if (S.saves   !== lastVals.sv) { pulseCell(vsv.parentElement); lastVals.sv = S.saves;   }

            vs.textContent = S.videos;
            vl.textContent = S.likes;
            vu.textContent = S.follows;
            vsh.textContent = S.shares;
            vsv.textContent = S.saves;
            const sec = S.watchSec + (S.running && S.videoStart ? (Date.now() - S.videoStart) / 1000 : 0);
            vw.textContent = sec < 60 ? `${sec.toFixed(0)}s` : `${(sec / 60).toFixed(1)}m`;
        } catch (e) {}
    }

    /* ========== KEYBINDS ========== */
    document.addEventListener('keydown', e => {
        try {
            if (e.ctrlKey && e.shiftKey && e.code === 'KeyS') { e.preventDefault(); S.running ? stop() : start(); }
            if (e.ctrlKey && e.shiftKey && e.code === 'KeyR') { e.preventDefault(); resetStats(); }
            if (e.ctrlKey && e.shiftKey && e.code === 'KeyD') {
                e.preventDefault();
                log('=== DUMP ===', '#0f0');
                console.log('[video]', findActiveVideo());
                console.log('[like]', findLikeBtn());
                console.log('[follow]', findFollowBtn());
                console.log('[share]', findShareBtn());
                console.log('[save]', findSaveBtn());
                console.log('[scroller]', findScrollTarget());
                const e2e = new Set();
                document.querySelectorAll('[data-e2e]').forEach(x => e2e.add(x.getAttribute('data-e2e')));
                console.log('[data-e2e]', [...e2e].sort());
            }
        } catch (err) {}
    }, true);

    /* ========== PUBLIC API ========== */
    window.__TT = {
        version: VERSION,
        start, stop, destroy, resetStats, next: doNext,
        like: doLike, follow: doFollow, share: doShare, save: doSave,
        swipe: () => simulateSwipeUp(findActiveVideo()),
        findSave: findSaveBtn,
        openPanel: () => S.ui && S.ui.openPanel(),
        closePanel: () => S.ui && S.ui.closePanel(),
        stats: () => ({
            version: VERSION,
            running: S.running, videos: S.videos, likes: S.likes, follows: S.follows,
            shares: S.shares, saves: S.saves,
            watchSec: +S.watchSec.toFixed(1),
            elapsed: ((Date.now() - S.startTs) / 1000).toFixed(0),
            rate: S.startTs ? (S.videos / Math.max((Date.now() - S.startTs) / 60000, 0.01)).toFixed(1) : '0',
            attachFail: S.attachFailCount,
            crashes: S.crashCount,
        }),
        config: CONFIG, toast,
        setSpeed: k => { if (CONFIG.speeds[k]) { CONFIG.speed = k; saveCfg(); updateUI(); toast(`Tốc độ: ${k}`, 'ok'); } },
        setHuman: on => { CONFIG.human.enabled = !!on; saveCfg(); toast(`Human: ${on ? 'ON' : 'OFF'}`, 'ok'); updateUI(); },
        setDelays: (like, follow, share, save) => {
            if (like)   CONFIG.likeDelay = like;
            if (follow) CONFIG.followDelay = follow;
            if (share)  CONFIG.shareDelay = share;
            if (save)   CONFIG.saveDelay = save;
            saveCfg();
            if (S.ui) { S.ui.drLike.refresh(); S.ui.drFollow.refresh(); S.ui.drShare.refresh(); S.ui.drSave.refresh(); }
            toast('Đã cập nhật delay', 'ok');
        },
        resetConfig: () => {
            for (const k in DEFAULTS) CONFIG[k] = DEFAULTS[k].slice();
            saveCfg();
            if (S.ui) { S.ui.drLike.refresh(); S.ui.drFollow.refresh(); S.ui.drShare.refresh(); S.ui.drSave.refresh(); }
            toast('↻ Đã reset config', 'ok');
        },
        resetFab: () => {
            try { localStorage.removeItem(FAB_POS_KEY); } catch (e) {}
            if (!S.ui) return;
            const f = S.ui.fab; f.style.left = ''; f.style.top = ''; f.style.right = '14px'; f.style.bottom = '90px';
        },
        exportConfig: () => {
            const out = {
                speed: CONFIG.speed, humanEnabled: CONFIG.human.enabled,
                autoNext: CONFIG.autoNext, autoLike: CONFIG.autoLike,
                autoFollow: CONFIG.autoFollow, autoShare: CONFIG.autoShare,
                autoSave: CONFIG.autoSave, autoUnmute: CONFIG.autoUnmute,
                likeDelay: CONFIG.likeDelay, followDelay: CONFIG.followDelay,
                shareDelay: CONFIG.shareDelay, saveDelay: CONFIG.saveDelay,
            };
            const j = JSON.stringify(out, null, 2);
            try { navigator.clipboard.writeText(j); } catch (e) {}
            console.log(j);
            return j;
        },
        importConfig: (obj) => {
            if (typeof obj === 'string') { try { obj = JSON.parse(obj); } catch (e) { toast('JSON lỗi', 'err'); return; } }
            if (!obj || typeof obj !== 'object') return;
            for (const k in obj) if (k in CONFIG) CONFIG[k] = obj[k];
            saveCfg();
            if (S.ui) { S.ui.drLike.refresh(); S.ui.drFollow.refresh(); S.ui.drShare.refresh(); S.ui.drSave.refresh(); }
            toast('✓ Import config', 'ok');
        },
    };

    /* ========== GLOBAL ERROR BOUNDARY ========== */
    const oldErr = window.onerror;
    window.onerror = function (msg, src, line, col, err) {
        if (S.running && /tt-auto|__TT/i.test(String(src) + String(msg))) {
            S.crashCount++;
            if (S.crashCount >= 10) recover('global errors');
        }
        if (oldErr) return oldErr.apply(this, arguments);
        return false;
    };

    /* ========== BOOT ========== */
    loadCfg();
    buildUI();
    S.heartbeat = setInterval(() => { try { if (S.running) { attachVideo(); pollVideo(); } } catch (e) { S.crashCount++; } }, 250);
    S.uiTick = setInterval(updateUI, 400);
    startWatchdog();
    startObserver();

    log(`v${VERSION} · sẵn sàng · human ${CONFIG.human.enabled ? 'ON' : 'OFF'}`, '#0f0');
})();
